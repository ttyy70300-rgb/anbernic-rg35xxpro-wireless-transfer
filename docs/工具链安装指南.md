## 一、为什么不用 WSL2

上一台开发电脑是 Windows 10 Enterprise LTSC，精简掉了 WSL 组件，所以走不通。
**这台机器不是 LTSC，WSL2 理论可用**，但实测有两个约束：

| 约束 | 影响 |
|---|---|
| 当前会话**非管理员** | 启用 `VirtualMachinePlatform` / WSL 组件需要管理员权限 + **重启** |
| `wsl.exe` **被本机安全策略拦截** | 即使装了也没法在当前环境里调用 |

所以直接走 Windows 原生路线。**好消息是：这条路完全够用**，而且不需要任何系统改动。

## 二、方案 A：Zig（推荐）

Zig 自带 musl libc，`zig cc` 一条命令就能交叉编译 aarch64，不需要单独装工具链、不需要 make。

### 2.1 安装（winget，一条命令）

```powershell
winget install --id zig.zig -e
```

装完验证：

```powershell
zig version
```

> 如果 winget 慢，直接去 https://ziglang.org/download/ 下 Windows x86_64 的 zip，
> 解压到 `C:\zig`，把 `C:\zig` 加进 PATH，效果一样。

### 2.2 验证交叉编译

新建一个测试文件：

```c
/* hello.c */
#include <stdio.h>

int main(void) {
    printf("Hello from aarch64!\n");
    return 0;
}
```

编译：

```powershell
zig cc -target aarch64-linux-musl -O2 -static hello.c -o hello_aarch64
```

检查产物架构（应该显示 `aarch64`）：

```powershell
C:\Users\Guo\.workbuddy\binaries\python\versions\3.13.12\python.exe -c "print(open('hello_aarch64','rb').read(20))"
```

> ELF 文件头第 4-5 字节是 `0xB7 0x00`，即 aarch64。

### 2.3 关键：让 CMake 用上 Zig

编译 SDL2 需要 CMake。装一个：

```powershell
winget install --id Kitware.CMake
```

然后在 CMake 调用里把编译器换成 `zig cc`：

```powershell
# 新建 build 目录时指定交叉编译工具链
zig cc -target aarch64-linux-musl $CMAKE_ARGS ...
```

实际做法（阶段 0-D 编译 SDL2 时用）：

```powershell
cmake -S . -B build-aarch64 `
  -DCMAKE_TOOLCHAIN_FILE=.\zig-aarch64.cmake `
  -DCMAKE_BUILD_TYPE=Release
```

配套的 `zig-aarch64.cmake`：

```cmake
set(CMAKE_SYSTEM_NAME Linux)
set(CMAKE_SYSTEM_PROCESSOR aarch64)

set(CMAKE_C_COMPILER   zig)
set(CMAKE_CXX_COMPILER zig)

# 把 zig 的目标三元组传给编译器
set(CMAKE_C_COMPILER_TARGET   aarch64-linux-musl)
set(CMAKE_CXX_COMPILER_TARGET aarch64-linux-musl)

set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM BEFORE)
set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
```

## 三、方案 B：aarch64-linux-musl-cross（备选）

不想用 Zig 就用这个 —— 传统 GCC 工具链，社区资料多，出问题好搜。

### 3.1 下载解压

- musl.cc：https://musl.cc/aarch64-linux-musl-cross.tgz
- 或 Bootlin：https://toolchains.bootlin.com/downloads/releases/toolchains/aarch64/tarballs/

下载后解压（本机有 `tar.exe`，Windows 11 自带）：

```powershell

cd tools\toolchain
tar -xzf aarch64-linux-musl-cross.tgz
```

得到：

```
│   ├── aarch64-linux-musl-gcc.exe
│   ├── aarch64-linux-musl-ld.exe
│   └── ...
├── aarch64-linux-musl/
│   ├── include/
│   └── lib/
└── ...
```

### 3.2 验证

```bash

aarch64-linux-musl-gcc --version
```

### 3.3 局限

**这个工具链不带 make 和 cmake**。所以：

- 编译单个 `.c` 文件 → 直接调 gcc，没问题
- 编译 SDL2 这种有构建系统的项目 → 需要先装 make（winget 装 Strawberry Perl 或
  `winget install GnuWin32.Make`），或改用 Zig 方案

## 五、阶段 0-B：零依赖 framebuffer 测试（下一步做什么）

装完工具链立刻做这个，用它验证工具链是否真的可用 —— 不依赖任何未知信息，最快证伪。

### 5.1 测试程序

`handheld/test/fbtest.c`：

```c
/* 零依赖 framebuffer 测试：直接在显存上画色块。
   目的：验证 aarch64 交叉编译器可用 + 掌机能执行我们编译的二进制。
   故意不依赖 SDL2，因为固件里的 SDL2 位置/版本还需 SSH 确认。 */

#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <sys/ioctl.h>
#include <sys/mman.h>
#include <unistd.h>
#include <linux/fb.h>

int main(void) {
    int fd = open("/dev/fb0", O_RDWR);
    if (fd < 0) {
        perror("open /dev/fb0");
        /* 即使没有 framebuffer 也要证明程序能跑 */
        printf("PocketTransfer fbtest: ran OK, but /dev/fb0 unavailable\n");
        return 0;
    }

    struct fb_var_screeninfo vinfo;
    struct fb_fix_screeninfo finfo;

    if (ioctl(fd, FBIOGET_VSCREENINFO, &vinfo) < 0 ||
        ioctl(fd, FBIOGET_FSCREENINFO, &finfo) < 0) {
        perror("ioctl FBIOGET");
        close(fd);
        return 1;
    }

    printf("fb: %ux%u  bpp=%u line_len=%u\n",
           vinfo.xres, vinfo.yres, vinfo.bits_per_pixel, finfo.line_length);

    size_t len = (size_t)finfo.line_length * vinfo.yres;
    unsigned char *fb = mmap(NULL, len, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (fb == MAP_FAILED) {
        perror("mmap");
        close(fd);
        return 1;
    }

    /* 三条横向色带：红 / 绿 / 蓝（RGB565） */
    const unsigned short bands[3] = { 0xF800, 0x07E0, 0x001F };
    unsigned short *px = (unsigned short *)fb;
    for (int y = 0; y < (int)vinfo.yres; y++) {
        unsigned short color = bands[y * 3 / vinfo.yres];
        for (int x = 0; x < (int)vinfo.xres; x++)
            px[y * (finfo.line_length / 2) + x] = color;
    }

    printf("PocketTransfer fbtest: OK, 3 color bands drawn\n");
    /* 不退出，留在屏幕上让用户肉眼确认；按 Ctrl+C 或杀进程结束 */
    sleep(3600);

    munmap(fb, len);
    close(fd);
    return 0;
}
```

编译：

```bash
# 方案 A（Zig）
zig cc -target aarch64-linux-musl -O2 -static test/fbtest.c -o build/fbtest

# 方案 B（musl-cross）
aarch64-linux-musl-gcc -O2 -static test/fbtest.c -o build/fbtest
```

### 5.2 部署与实测

1. 拷 `fbtest` 到 TF1 的 `Roms/APPS/PocketTransfer.sh` 同级（即 `Roms/APPS/` 下）
2. 写一个最小 `fbtest.sh`：

```bash
#!/bin/bash
progdir="$(cd "$(dirname "$0")"; pwd)"
basedir="$(cd "$progdir/../.."; pwd)"
cd "$progdir/PocketTransfer" || exit 1
mkdir -p logs
exec >logs/stdout.log 2>&1
export BASE_PATH="$basedir"
./fbtest
exit 0
```

3. 插卡开机 → Apps Center → 点开
4. 屏幕上应出现红/绿/蓝三条横带
5. `logs/stdout.log` 里应有 `fb: 640x480 bpp=16 ...` 的输出

**这一步通了 → 工具链路线确认可行，阶段 0 完成一半。**
